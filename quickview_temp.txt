import React, { useState, useEffect, useRef } from 'react';
import { motion, AnimatePresence } from 'motion/react';
import { useApp } from '../context/AppContext.js';
import {
  X,
  ShoppingBag,
  Heart,
  Star,
  Truck,
  RotateCcw,
  Sparkles,
  ChevronRight,
  ChevronLeft,
  Layers,
  ZoomIn,
  Share2,
} from 'lucide-react';

const VIEW_TITLES_AR = [
  'الواجهة الأمامية للمنتج',
  'الجهة الخلفية للمنتج',
  'عرض تفاصيل الخامة والقماش',
  'التفاصيل الدقيقة ونوع القطن',
];

const VIEW_TITLES_EN = [
  'Front View (Design & Silhouette)',
  'Back View (Rear Fit & Seams)',
  'Fabric & Texture Close-Up',
  'Fine Details & Luxury Stitching',
];

export const QuickViewModal: React.FC = () => {
  const {
    quickViewProduct,
    setQuickViewProduct,
    language,
    theme,
    addToCart,
    toggleWishlist,
    isInWishlist,
  } = useApp();

  const isAr = language === 'ar';
  const product = quickViewProduct;
  const isWish = product ? isInWishlist(product.id) : false;

  const [activeSlot, setActiveSlot] = useState<number>(0);
  const [direction, setDirection] = useState<number>(0);
  const [selectedSize, setSelectedSize] = useState<string>('Standard');
  const [selectedColor, setSelectedColor] = useState<string>('#120205');
  const [quantity, setQuantity] = useState(1);
  const [showSizeGuide, setShowSizeGuide] = useState(false);
  const [isZooming, setIsZooming] = useState(false);
  const [zoomOrigin, setZoomOrigin] = useState({ x: '50%', y: '50%' });

  // Touch swipe handling
  const touchStartX = useRef<number | null>(null);
  const touchStartY = useRef<number | null>(null);

  // Sync selection when product changes
  useEffect(() => {
    if (quickViewProduct) {
      setActiveSlot(0);
      setDirection(0);
      setSelectedSize(quickViewProduct.sizes?.[0] || 'Standard');
      setSelectedColor(quickViewProduct.colors?.[0] || '#120205');
      setQuantity(1);
    }
  }, [quickViewProduct]);

  if (!quickViewProduct || !product) return null;

  const productImages: string[] =
    product.images && product.images.length === 4
      ? product.images
      : [product.image, product.image, product.image, product.image];

  const handleAddToCart = () => {
    if (!product.inStock) return;
    addToCart(product, selectedSize, selectedColor, quantity);
    setQuickViewProduct(null);
  };

  const handlePrevImage = () => {
    setDirection(-1);
    setActiveSlot((prev) => (prev === 0 ? productImages.length - 1 : prev - 1));
  };

  const handleNextImage = () => {
    setDirection(1);
    setActiveSlot((prev) => (prev === productImages.length - 1 ? 0 : prev + 1));
  };

  const handleDragEnd = (e: any, info: any) => {
    const swipeThreshold = 30;
    const { offset, velocity } = info;

    if (offset.x < -swipeThreshold || velocity.x < -500) {
      handleNextImage();
    } else if (offset.x > swipeThreshold || velocity.x > 500) {
      handlePrevImage();
    }
  };

  const handleShare = async () => {
    try {
      const shareData = {
        title: isAr ? product.title : product.titleEn,
        text: isAr ? product.description : product.descriptionEn,
        url: window.location.href,
      };
      if (navigator.share) {
        await navigator.share(shareData);
      } else {
        await navigator.clipboard.writeText(window.location.href);
        alert(isAr ? 'تم نسخ الرابط!' : 'Link copied to clipboard!');
      }
    } catch (error) {
      console.log('Error sharing:', error);
    }
  };

  return (
    <div
      id="quickview-modal-backdrop"
      onClick={() => setQuickViewProduct(null)}
      className="fixed inset-0 z-50 flex items-end sm:items-center justify-center p-0 sm:p-4 bg-black/50 backdrop-blur-sm animate-fade-in"
    >
      {/* Bottom sheet on mobile, rounded card on desktop */}
      <div
        id="quickview-modal-content"
        onClick={(e) => e.stopPropagation()}
        className="w-full sm:max-w-4xl max-h-[92vh] overflow-y-auto bg-white text-[#111111] border-t sm:border border-neutral-200 rounded-t-3xl sm:rounded-3xl shadow-2xl p-5 sm:p-7 animate-slide-up-sheet sm:animate-scale-in relative no-scrollbar"
      >
        {/* Mobile Pull Indicator */}
        <div className="sm:hidden w-full flex justify-center pb-3">
          <div className="w-12 h-1.5 bg-neutral-700 rounded-full" />
        </div>

        {/* Top Actions */}
        <div className="absolute top-4 right-4 rtl:right-auto rtl:left-4 flex items-center gap-2 z-20">
          <button
            onClick={handleShare}
            className="w-9 h-9 rounded-full bg-white hover:bg-neutral-100 flex items-center justify-center text-neutral-500 hover:text-[#111111] transition-colors cursor-pointer border border-neutral-200"
            title={isAr ? 'مشاركة' : 'Share'}
          >
            <Share2 className="w-4 h-4" />
          </button>
          <button
            id="btn-close-quickview"
            onClick={() => setQuickViewProduct(null)}
            className="w-9 h-9 rounded-full bg-white hover:bg-neutral-100 flex items-center justify-center text-neutral-500 hover:text-[#111111] transition-colors cursor-pointer border border-neutral-200"
          >
            <X className="w-5 h-5" />
          </button>
        </div>

        <div className="grid grid-cols-1 lg:grid-cols-12 gap-5 lg:gap-7 items-start">
          {/* 1. 4-Image Interactive Gallery Column (5 cols on lg) */}
          <div className="lg:col-span-6 space-y-3">
            {/* Main Active Image Box */}
            <motion.div
              className="relative aspect-square rounded-2xl overflow-hidden p-4 flex items-center justify-center border border-neutral-200 bg-neutral-50 select-none touch-pan-y"
              drag="x"
              dragConstraints={{ left: 0, right: 0 }}
              dragElastic={0.2}
              onDragEnd={handleDragEnd}
            >
              {/* Product Badge */}
              {product.badge && (
                <span className="absolute top-3 right-3 rtl:right-auto rtl:left-3 px-3 py-1 rounded-full text-xs font-bold bg-[#111111] text-white border border-[#5A0A19] shadow-md flex items-center gap-1 z-10">
                  <Sparkles className="w-3 h-3 text-rose-300" />
                  {product.badge}
                </span>
              )}

              {/* View Slot Title Pill */}
              <div className="absolute top-3 left-3 rtl:left-auto rtl:right-3 z-10">
                <span className="px-3 py-1 rounded-full text-[10px] font-bold bg-black/50 backdrop-blur-md text-[#111111] border border-neutral-200 flex items-center gap-1.5 shadow-md">
                  <Layers className="w-3 h-3 text-rose-300" />
                  <span>
                    {isAr ? VIEW_TITLES_AR[activeSlot] : VIEW_TITLES_EN[activeSlot]} ({activeSlot + 1}/4)
                  </span>
                </span>
              </div>

              {/* Arrow Nav Buttons */}
              <button
                onClick={handlePrevImage}
                className="absolute left-2 top-1/2 -translate-y-1/2 w-8 h-8 rounded-full bg-black/70 hover:bg-black text-[#111111] flex items-center justify-center transition-all z-10 cursor-pointer shadow-lg border border-neutral-200"
                title={isAr ? 'الصورة السابقة' : 'Previous'}
              >
                <ChevronLeft className="w-4 h-4" />
              </button>

              <button
                onClick={handleNextImage}
                className="absolute right-2 top-1/2 -translate-y-1/2 w-8 h-8 rounded-full bg-black/70 hover:bg-black text-[#111111] flex items-center justify-center transition-all z-10 cursor-pointer shadow-lg border border-neutral-200"
                title={isAr ? 'الصورة التالية' : 'Next'}
              >
                <ChevronRight className="w-4 h-4" />
              </button>

              {/* Main Image Stack with animated motion slide */}
              <div 
                className="w-full h-full relative overflow-hidden flex items-center justify-center cursor-zoom-in"
                onMouseEnter={() => setIsZooming(true)}
                onMouseLeave={() => { setIsZooming(false); setZoomOrigin({ x: '50%', y: '50%' }); }}
                onMouseMove={(e) => {
                  const { left, top, width, height } = e.currentTarget.getBoundingClientRect();
                  const x = ((e.clientX - left) / width) * 100;
                  const y = ((e.clientY - top) / height) * 100;
                  setZoomOrigin({ x: `${x}%`, y: `${y}%` });
                }}
              >
                <AnimatePresence initial={false} custom={direction} mode="wait">
                  <motion.img
                    key={activeSlot}
                    src={productImages[activeSlot]}
                    alt={`${isAr ? product.title : product.titleEn} - View ${activeSlot + 1}`}
                    initial={{ opacity: 0, x: direction > 0 ? 50 : -50, scale: 1 }}
                    animate={{ opacity: 1, x: 0, scale: isZooming ? 2.2 : 1 }}
                    exit={{ opacity: 0, x: direction > 0 ? -50 : 50, scale: 1 }}
                    transition={{ duration: 0.25, ease: 'easeOut' }}
                    className="absolute inset-0 w-full h-full object-contain p-2"
                    style={{
                      transformOrigin: `${zoomOrigin.x} ${zoomOrigin.y}`
                    }}
                  />
                </AnimatePresence>
              </div>
            </motion.div>

            {/* 4 Micro Thumbnails Switcher */}
            <div className="grid grid-cols-4 gap-2">
              {productImages.map((thumbSrc, idx) => (
                <button
                  key={idx}
                  onClick={() => {
                    setDirection(idx > activeSlot ? 1 : -1);
                    setActiveSlot(idx);
                  }}
                  className={`relative rounded-xl overflow-hidden p-1 bg-neutral-50 border transition-all cursor-pointer flex flex-col items-center gap-1 ${
                    activeSlot === idx
                      ? 'border-[#120205] ring-2 ring-[#120205]/60 bg-[#221014]'
                      : 'border-neutral-200 opacity-70 hover:opacity-100 hover:border-neutral-500'
                  }`}
                >
                  <div className="w-full aspect-square relative rounded-lg overflow-hidden flex items-center justify-center">
                    <img
                      src={thumbSrc}
                      alt={`Thumb ${idx + 1}`}
                      className="w-full h-full object-contain p-1"
                    />
                  </div>
                  <span className="text-[9px] font-bold text-center truncate w-full px-0.5 text-neutral-600">
                    {idx === 0
                      ? isAr
                        ? '1. الأمامية'
                        : '1. Front'
                      : idx === 1
                      ? isAr
                        ? '2. الخلفية'
                        : '2. Back'
                      : idx === 2
                      ? isAr
                        ? '3. القماش'
                        : '3. Fabric'
                      : isAr
                      ? '4. التطريز'
                      : '4. Detail'}
                  </span>
                </button>
              ))}
            </div>
          </div>

          {/* 2. Product Meta & Actions Column (6 cols on lg) */}
          <div className="lg:col-span-6 space-y-4">
            <div>
              <div className="flex items-center justify-between">
                <span className="text-[10px] text-neutral-500 font-mono uppercase tracking-wider">{product.sku}</span>
                <span className="text-[10px] px-2 py-0.5 rounded-full bg-neutral-800 text-neutral-500 font-mono">
                  {product.category}
                </span>
              </div>
              <h2 className="font-luxury font-bold text-xl sm:text-2xl mt-1 text-[#111111] leading-tight">
                {isAr ? product.title : product.titleEn}
              </h2>
            </div>

            {/* Price & Rating */}
            <div className="flex items-center justify-between">
              <div className="flex items-baseline gap-2">
                <span className="font-mono font-bold text-xl text-white bg-[#111111] px-3.5 py-1 rounded-xl border border-[#5A0A19] shadow-md">
                  {product.price} {isAr ? 'د.أ' : 'JOD'}
                </span>
                {product.oldPrice && (
                  <span className="text-sm text-neutral-500 line-through font-mono">
                    {product.oldPrice} {isAr ? 'د.أ' : 'JOD'}
                  </span>
                )}
              </div>

              <div className="flex items-center gap-1.5 text-amber-400 text-xs font-bold font-mono bg-[#181818] px-3 py-1.5 rounded-xl border border-neutral-200">
                <Star className="w-4 h-4 fill-amber-400" />
                <span>{product.rating}</span>
                <span className="text-neutral-500 font-normal">({product.reviewsCount} تقييم)</span>
              </div>
            </div>

            {/* Description */}
            <p className="text-xs sm:text-sm text-neutral-600 leading-relaxed bg-neutral-50 p-3.5 rounded-xl border border-neutral-200">
              {isAr ? product.description : product.descriptionEn}
            </p>

            {/* Sizes Selection */}
            {product.sizes && product.sizes.length > 0 && (
              <div className="space-y-1.5">
                <div className="flex justify-between items-end mb-1">
                  <div className="text-xs font-bold uppercase text-neutral-500 font-mono">
                    {isAr ? 'اختر المقاس:' : 'Select Size:'}
                  </div>
                  <button 
                    onClick={() => setShowSizeGuide(true)} 
                    className="text-[10px] text-rose-500 font-bold underline cursor-pointer hover:text-rose-400"
                  >
                     {isAr ? 'دليل المقاسات' : 'Size Guide'}
                  </button>
                </div>
                <div className="flex flex-wrap gap-2">
                  {product.sizes.map((size) => (
                    <button
                      key={size}
                      onClick={() => setSelectedSize(size)}
                      className={`px-3.5 py-1.5 rounded-xl text-xs font-mono font-semibold transition-all cursor-pointer border ${
                        selectedSize === size
                          ? 'bg-[#111111] text-white border-[#5A0A19] shadow-md ring-1 ring-[#120205]'
                          : 'bg-neutral-50 text-neutral-600 border-[#2A2A2A] hover:border-neutral-500'
                      }`}
                    >
                      {size}
                    </button>
                  ))}
                </div>
              </div>
            )}

            {/* Quantity Selector & Wishlist */}
            <div className="flex items-center justify-between pt-1">
              <div className="flex items-center gap-3">
                <span className="text-xs font-bold uppercase text-neutral-500 font-mono">
                  {isAr ? 'الكمية:' : 'Qty:'}
                </span>
                <div className="flex items-center rounded-xl border border-neutral-200 bg-neutral-50">
                  <button
                    onClick={() => setQuantity((q) => Math.max(1, q - 1))}
                    className="px-3.5 py-1.5 hover:bg-[#252525] text-xs font-bold text-neutral-600 cursor-pointer"
                  >
                    -
                  </button>
                  <span className="px-3.5 py-1.5 text-xs font-mono font-bold text-[#111111]">{quantity}</span>
                  <button
                    onClick={() => setQuantity((q) => q + 1)}
                    className="px-3.5 py-1.5 hover:bg-[#252525] text-xs font-bold text-neutral-600 cursor-pointer"
                  >
                    +
                  </button>
                </div>
              </div>

              <button
                onClick={() => toggleWishlist(product.id)}
                className={`p-2.5 rounded-xl border transition-all cursor-pointer ${
                  isWish
                    ? 'text-white bg-[#800020] border-[#5A0A19] shadow-xs'
                    : 'text-neutral-500 hover:text-[#111111] bg-neutral-50 border-neutral-200 hover:border-[#800020]'
                }`}
                title={isAr ? 'المفضلة' : 'Wishlist'}
              >
                <Heart className={`w-4 h-4 ${isWish ? 'fill-white text-white' : ''}`} />
              </button>
            </div>

            {/* Add to Bag Action */}
            <div className="pt-2">
              <button
                id="btn-modal-add-to-cart"
                onClick={handleAddToCart}
                disabled={!product.inStock}
                className="w-full py-3.5 px-6 rounded-xl bg-[#111111] hover:bg-[#333333] text-white font-bold text-xs uppercase tracking-wider transition-all flex items-center justify-center gap-2 shadow-xl border border-[#5A0A19] disabled:opacity-50 cursor-pointer"
              >
                <ShoppingBag className="w-4 h-4" />
                <span>
                  {!product.inStock
                    ? isAr
                      ? 'الكمية غير متوفرة حالياً'
                      : 'Sold Out'
                    : isAr
                    ? 'أضف للسلة يا كشخة'
                    : 'Add to Bag'}
                </span>
              </button>
            </div>

            {/* Jordan Guarantees */}
            <div className="pt-3 border-t border-neutral-200 grid grid-cols-2 gap-2 text-[11px] text-neutral-500">
              <div className="flex items-center gap-1.5">
                <Truck className="w-3.5 h-3.5 text-rose-400 shrink-0" />
                <span>{isAr ? 'دينار لعمان / دينارين للمحافظات' : '1 JOD Amman / 2 JOD Jordan'}</span>
              </div>
              <div className="flex items-center gap-1.5">
                <RotateCcw className="w-3.5 h-3.5 text-rose-400 shrink-0" />
                <span>{isAr ? 'تبديل فوري وسهل' : '3 Days Easy Return'}</span>
              </div>
            </div>
          </div>
        </div>
      </div>

      {/* Size Guide Overlay */}
      <AnimatePresence>
        {showSizeGuide && (
          <motion.div
            initial={{ opacity: 0 }}
            animate={{ opacity: 1 }}
            exit={{ opacity: 0 }}
            className="absolute inset-0 z-[60] flex items-center justify-center p-4 bg-black/60 backdrop-blur-md"
            onClick={(e) => {
              e.stopPropagation();
              setShowSizeGuide(false);
            }}
          >
            <div 
              onClick={(e) => e.stopPropagation()} 
              className="bg-white rounded-2xl w-full max-w-sm p-5 relative shadow-2xl border border-neutral-200"
            >
              <button
                onClick={() => setShowSizeGuide(false)}
                className="absolute top-3 right-3 rtl:right-auto rtl:left-3 w-8 h-8 rounded-full bg-neutral-100 flex items-center justify-center text-neutral-500 hover:text-black cursor-pointer transition-colors"
              >
                <X className="w-4 h-4" />
              </button>
              <h3 className="font-bold text-lg mb-4 text-[#111111]">{isAr ? 'دليل المقاسات' : 'Size Guide'}</h3>
              
              <div className="overflow-x-auto rounded-xl border border-neutral-200">
                <table className="w-full text-xs text-left rtl:text-right text-[#111111]">
                  <thead className="bg-neutral-100 text-neutral-600 border-b border-neutral-200">
                    <tr>
                      <th className="px-3 py-2 font-bold">{isAr ? 'المقاس' : 'Size'}</th>
                      <th className="px-3 py-2 font-bold border-l border-neutral-200">{isAr ? 'الصدر (سم)' : 'Chest (cm)'}</th>
                      <th className="px-3 py-2 font-bold border-l border-neutral-200">{isAr ? 'الطول (سم)' : 'Length (cm)'}</th>
                    </tr>
                  </thead>
                  <tbody className="divide-y divide-neutral-200">
                    <tr className="bg-white">
                      <td className="px-3 py-2 font-bold">S</td>
                      <td className="px-3 py-2 border-l border-neutral-200">50</td>
                      <td className="px-3 py-2 border-l border-neutral-200">70</td>
                    </tr>
                    <tr className="bg-neutral-50">
                      <td className="px-3 py-2 font-bold">M</td>
                      <td className="px-3 py-2 border-l border-neutral-200">53</td>
                      <td className="px-3 py-2 border-l border-neutral-200">72</td>
                    </tr>
                    <tr className="bg-white">
                      <td className="px-3 py-2 font-bold">L</td>
                      <td className="px-3 py-2 border-l border-neutral-200">56</td>
                      <td className="px-3 py-2 border-l border-neutral-200">74</td>
                    </tr>
                    <tr className="bg-neutral-50">
                      <td className="px-3 py-2 font-bold">XL</td>
                      <td className="px-3 py-2 border-l border-neutral-200">59</td>
                      <td className="px-3 py-2 border-l border-neutral-200">76</td>
                    </tr>
                    <tr className="bg-white">
                      <td className="px-3 py-2 font-bold">XXL</td>
                      <td className="px-3 py-2 border-l border-neutral-200">62</td>
                      <td className="px-3 py-2 border-l border-neutral-200">78</td>
                    </tr>
                  </tbody>
                </table>
              </div>
              
              <p className="text-[10px] text-neutral-500 mt-4 leading-relaxed">
                {isAr 
                  ? '* المقاسات تقريبية وقد تختلف بحدود 1-2 سم حسب التصميم. ينصح بقياس قطعة ملابس مشابهة تملكها لمقارنة المقاسات بشكل أفضل.' 
                  : '* Measurements are approximate and may vary by 1-2 cm depending on the design. We recommend measuring a similar garment you own to compare.'}
              </p>
            </div>
          </motion.div>
        )}
      </AnimatePresence>
    </div>
  );
};
